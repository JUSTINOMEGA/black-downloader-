from flask import Flask, request, jsonify, render_template from flask_cors import CORS import yt_dlp import os app = Flask(__name__) CORS(app) @app.route('/') def home(): return render_template('index.html') @app.route('/api/download') def download(): video_url = request.args.get('url') if not video_url: return jsonify({"error": "No URL"}), 400 # Options for yt-dlp ydl_opts = { 'format': 'best', 'quiet': True, 'no_warnings': True, 'outtmpl': '%(title)s.%(ext)s', } try: with yt_dlp.YoutubeDL(ydl_opts) as ydl: info = ydl.extract_info(video_url, download=False) # Get best video URL download_url = info.get('url') if not download_url and 'entries' in info: download_url = info['entries'][0].get('url') # For TikTok, get direct URL if not download_url: download_url = info.get('requested_formats', [{}])[0].get('url') or info.get('formats', [{}])[-1].get('url') return jsonify({ "downloadUrl": download_url, "title": info.get('title', 'video') }) except Exception as e: print(f"Error: {e}") return jsonify({"error": str(e)}), 500 if __name__ == '__main__': port = int(os.environ.get("PORT", 10000)) app.run(host='0.0.0.0', port=port)
